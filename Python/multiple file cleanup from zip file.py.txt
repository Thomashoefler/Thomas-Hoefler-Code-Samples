# LOGGING *****************************************************************************************************************

class logger:
    def __init__(self, logDataFile):
        self.logDataFile = logDataFile
        self.logTime = datetime.now().strftime("%Y%m%d-%H%M%S")
        self.logName = "Files_Cleanup_{}_{}.log".format(self.logDataFile, self.logTime)

    def log(self, label, message):
        if label != '' and message != '':
            logline = "{} [{}]: {}".format(datetime.now(),label.upper(),message)
            print(logline)
            with open (self.logName, 'a') as lg:
                lg.write(logline+"\n")
        else:
            print("No message given!")

    def clean(self):
        if os.path.exists(self.logName):
            try:
                os.remove(self.logName)
            except OSError:
                print("Failed to remove the logfile.log")
            else:
                print("Successfully removed the logfile.log")
